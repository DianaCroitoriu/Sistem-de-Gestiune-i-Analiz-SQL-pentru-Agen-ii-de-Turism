# Turism2.7 Pentru fiecare angajat, afișați, pe coloane separate, 
# încăsările (rezervărilor pentru care au fost persoane de contact) 
# pentru anii 2018, 2019 și 2020. 
turism_2_7 <- persoane %>%
  inner_join(angajati, by = c("idpersoana" = "idangajat")) %>%
  inner_join(rezervari, by = c("idpersoana" = "idpersoanacontact")) %>%
  inner_join(incasari_rezervari, by = "idrezervare") %>%
  inner_join(incasari, by = "idincasare") %>%
  group_by(numepersoana, prenumepersoana) %>%
  summarise(
    incasari2018 = sum(if_else(year(dataoraincasari) == 2018, transa_rezervare, 0)),
    incasari2019 = sum(if_else(year(dataoraincasari) == 2019, transa_rezervare, 0)),
    incasari2020 = sum(if_else(year(dataoraincasari) == 2020, transa_rezervare, 0))
  )


# Turism2.8 Care sunt angajații noștri care nu au acordat clienților lor 
# niciun discount (până în prezent)?

# Turism2.13 Care sunt clienții care s-au cazat în (cel puțin) 
# toate hotelurile în care s-a cazat clientul Ionescu? 

turism_2_13 <- persoane %>%
  filter(numepersoana != "Ionescu") %>%
  inner_join(rezervari, by = c("idpersoana" = "codclient")) %>%
  inner_join(rezervari_sejururi, by = "idrezervare") %>%
  inner_join(sejururi, by = "idsejur") %>%
  inner_join(camere_hotel, by = "idcamera") %>%
  group_by(idpersoana, numepersoana, prenumepersoana) %>%

  summarise(
    hoteluri_intersectie = n_distinct(idhotel[idhotel %in% (
      persoane %>%
        inner_join(rezervari, by = c("idpersoana" = "codclient")) %>%
        inner_join(rezervari_sejururi, by = "idrezervare") %>%
        inner_join(sejururi, by = "idsejur") %>%
        inner_join(camere_hotel, by = "idcamera") %>%
        filter(numepersoana == "Ionescu") %>%
        distinct(idhotel) %>%
        pull(idhotel)
    )]),
    total_hoteluri_ionescu = n_distinct(
      persoane %>%
        inner_join(rezervari, by = c("idpersoana" = "codclient")) %>%
        inner_join(rezervari_sejururi, by = "idrezervare") %>%
        inner_join(sejururi, by = "idsejur") %>%
        inner_join(camere_hotel, by = "idcamera") %>%
        filter(numepersoana == "Ionescu") %>%
        pull(idhotel)
    ),
    .groups = "drop"
  ) %>%
  filter(hoteluri_intersectie == total_hoteluri_ionescu) %>%
  select(idpersoana, numepersoana, prenumepersoana)


# Turism2.14 Calculați ponderea clienților din România 
# în totalul încasărilor hotelurilor din Italia în 2023.

turism_2_14 <- incasari %>%
  inner_join(incasari_rezervari, by = "idincasare") %>%
  inner_join(rezervari, by = "idrezervare") %>%
  inner_join(persoane, by = c("codclient" = "idpersoana")) %>%
  inner_join(localitati, by = c("idlocalitatepersoana" = "idlocalitate")) %>% # Localitate persoana
  inner_join(tari, by = "idtara") %>% # Tara persoana
  inner_join(rezervari_sejururi, by = "idrezervare") %>%
  inner_join(sejururi, by = "idsejur") %>%
  inner_join(camere_hotel, by = "idcamera") %>%
  inner_join(hoteluri, by = "idhotel") %>%
  inner_join(localitati, by = c("idlocalitate" = "idlocalitate"), suffix = c("_p", "_h")) %>% # Localitate hotel
  inner_join(tari, by = c("idtara_h" = "idtara"), suffix = c("_p", "_h")) %>% # Tara hotel
  
  filter(
    numetara_h == "Italia",
    year(dataoraincasari) == 2023
  ) %>%
  
  summarise(
    ponderea = (sum(transa_rezervare[numetara_p == "Romania"], na.rm = TRUE) / 
                  sum(transa_rezervare, na.rm = TRUE)) * 100
  ) 


# Turism2.15 Pe ce poziție se găsește hotelul X în 
# topul hotelurilor din Grecia, după numărul de sejururi?

turism_2_15 <- sejururi %>%
  inner_join(camere_hotel, by = "idcamera") %>%
  inner_join(hoteluri, by = "idhotel") %>%
  inner_join(localitati, by = "idlocalitate") %>%
  inner_join(tari, by = "idtara") %>%
  filter(numetara == "Grecia") %>%
  
  group_by(numehotel) %>%
  summarise(nr_sejururi = n(), .groups = "drop") %>%
  mutate(pozitie = min_rank(-nr_sejururi)) %>%
  filter(numehotel == "Hotel X") %>%
  
  select(numehotel, nr_sejururi, pozitie)
